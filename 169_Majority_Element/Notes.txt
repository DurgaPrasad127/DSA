class Solution:
    def majorityElement(self, nums: list[int]) -> int:
        maj = len(nums) // 2

        for i in range(len(nums)):
            maj_ele = 0
            for j in range(len(nums)):
                if nums[i] == nums[j]:
                    maj_ele += 1

            if maj_ele > maj:
                return nums[i]

class Solution:
    def majorityElement(self, nums: list[int]) -> int:
        maj = len(nums) // 2
        freq = {}

        for num in nums:
            if num in freq:
                freq[num] += 1
            else:
                freq[num] = 1

            if freq[num] > maj:
                return num

    count = 0

    for num in nums:
        if count == 0:
            candidate = num

        if num == candidate:
