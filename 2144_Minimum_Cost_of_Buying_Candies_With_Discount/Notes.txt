class Solution:
    def minimumCost(self, cost: list[int]) -> int:
        cost.sort()
        n=len(cost)
        ans=0
        took=0
        for i in range(n-1,-1,-1):
            if took==2:
                took=0
            else:
                ans+=cost[i]
                took+=1
        return ans

class Solution:
    def minimumCost(self, cost: list[int]) -> int:
        if len(cost) == 2:
            return cost[0] + cost[1]
        if len(cost) == 1:
            return cost[0]
        
        cost.sort(reverse=True)
        mn = 0
        for i in range(len(cost)):
            if i % 3 != 2:
                mn += cost[i]
        return mn
