class Solution:
    def minimumSumSubarray(self, nums: List[int], l: int, r: int) -> int:
        res = []

        for i in range(len(nums)):
            for j in range(l, r + 1):
                if i + j <= len(nums):
                    s = sum(nums[i:i+j])

                    if s > 0:
                        res.append(s)

        if not res:
            return -1

        return min(res)
